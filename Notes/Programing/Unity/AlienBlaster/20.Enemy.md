### `InvokeRepeating`
pass method, start delay, interval between calls.
```cs
using UnityEngine;

public class Frog : MonoBehaviour {
    [SerializeField] float _jumpDelay = 3;
    [SerializeField] Vector2 _jumpForce;

    Rigidbody2D _rb;
    SpriteRenderer _sr;

    void Awake() {
        _rb = GetComponent<Rigidbody2D>();
        _sr = GetComponent<SpriteRenderer>();
        InvokeRepeating("Jump", _jumpDelay, _jumpDelay);
    }
    void Jump() {
        _rb.AddForce(_jumpForce);
        _jumpForce *= new Vector2(-1, 1); // side change(flip x)
    }
}
```

```cs
public class Frog : MonoBehaviour {
    [SerializeField] float _jumpDelay = 3;
    [SerializeField] Vector2 _jumpForce;
    [SerializeField] Sprite _jumpingSprite;
    [SerializeField] int _Jumps = 3;

    Rigidbody2D _rb;
    SpriteRenderer _sr;
    Sprite _original;
    private int _jumpRemaining;

    void Awake() {
        _rb = GetComponent<Rigidbody2D>();
        _sr = GetComponent<SpriteRenderer>();
        _original = _sr.sprite;
        InvokeRepeating("Jump", _jumpDelay, _jumpDelay);
        _jumpRemaining = _Jumps;
    }
    void OnCollisionEnter2D(Collision2D collision) {
        _sr.sprite = _original;
    }
    void Jump() {
        if(_jumpRemaining <= 0) {
            _jumpForce *= new Vector2(-1, 1);
            _jumpRemaining=_Jumps;
        }
        _sr.sprite = _jumpingSprite;
        _sr.flipX = _jumpForce.x > 0;
        _jumpRemaining--;
        _rb.AddForce(_jumpForce);
    }
}
```
