        {
            _jumpEndTime = Time.time + _jumpTime;
            _jumpsAvailable--;
            _audioSource.pitch = (_jumpsAvailable > 0) ? 1 : 1.3f;
            _audioSource.Play();
        }
        if (_playerInput.actions["Jump"].ReadValue<float>()>0 && Time.time < _jumpEndTime)
        {
            vertical = _jumpVelocity;
        }
        var desiredVeclocity = horizontalInput * _maxHorizonatalSpeed;
        var acceleration = _isOnSnow ? _snowAcceleration : _grassAcceleration;
        _horizontal = Mathf.Lerp(_horizontal, desiredVeclocity, Time.deltaTime * acceleration);

        UpdateSprite();
    }
