```java
@RestController
@RequestMapping("/api")
public class SummarizeController {
    private final SummarizeService summarizeService;

    public SummarizeController(SummarizeService summarizeService) {
        this.summarizeService = summarizeService;
    }

    @PostMapping("/summarize")
    public String summarize(@RequestBody String request) {
        if (request == null) {
            throw new ResponseStatusException(HttpStatus.BAD_REQUEST, "Field 'text' is required");
        }
        return summarizeService.summarize(request);
    }

    @PostMapping("/chat")
    public String chat(@RequestBody String request) {
        return summarizeService.chat(request);
    }
}
```
for service
```java
@Service
public class SummarizeService {

    private final ChatClient chatClient;

    public SummarizeService(ChatClient.Builder chatClientBuilder) {
        this.chatClient = chatClientBuilder.build();
    }

    public String summarize(String text) {
        return chatClient.prompt()
            .system("Summarize the user's text in a few concise sentences.")
            .user(text)
            .call()
            .content();
    }

    public String chat(String message) {
        return chatClient.prompt()
            .user(message)
            .call()
            .content();
    }
}
```
