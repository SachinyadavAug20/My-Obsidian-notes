### Spring code
to change sprint when some thing is on it and contract when nothing.
add collision methods => we get collision call back when something will collides with each other.

```cs
using UnityEngine;

public class Spring : MonoBehaviour {
    [SerializeField] Sprite _sprung;

    SpriteRenderer _sr;
    void Awake() {
        _sr = GetComponent<SpriteRenderer>();
    }
    private void OnCollisionEnter2D(Collision2D collision) {
        if (collision.gameObject.CompareTag("Player")) {
            _sr.sprite = _sprung;
        }
    }
}
```

```cs
using UnityEngine;

public class Spring : MonoBehaviour {
    [SerializeField] Sprite _sprung;
    Sprite _original;

    SpriteRenderer _sr;
    void Awake() {
        _sr = GetComponent<SpriteRenderer>();
        _original=_sr.sprite;
    }
    private void OnCollisionEnter2D(Collision2D collision) {
        if (collision.gameObject.CompareTag("Player")) {
            _sr.sprite = _sprung;
        }
    }
    private void OnCollisionExit2D(Collision2D collision) {
        if (collision.gameObject.CompareTag("Player")) {
            _sr.sprite = _original;
        }
    }
}
```
