```cs
using UnityEngine;

public class Brick : MonoBehaviour {
    [SerializeField] ParticleSystem _brickParticle;
    [SerializeField] float _laserDestructionTime=1f;
    float _takenDamageTime;
    void OnCollisionEnter2D(Collision2D collision) {
        var player=collision.gameObject.GetComponent<Player>();
        if(player==null)return;
        Vector2 normal=collision.contacts[0].normal;
        float dot=Vector2.Dot(normal,Vector2.up);
        Debug.Log("dot="+dot);
        if(dot>0.5){
            player.StopJump();
            Explod();
        }
    }
    void Explod(){
        Instantiate(_brickParticle,transform.position,Quaternion.identity);// pass rotation
        Destroy(gameObject);
    }
    public void TakeLaserDamager(){
        _takenDamageTime+=Time.deltaTime;
        if(_takenDamageTime>=_laserDestructionTime) Explod();
    }
}
```
