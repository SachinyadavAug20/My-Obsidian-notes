using UnityEngine;

public class Ladybug : MonoBehaviour
{
    Rigidbody2D _rb;
    SpriteRenderer _sr;
    Collider2D _col;
    [SerializeField] float _speed = 1;
    [SerializeField] float _rayCastDistance = 0.2f;
    Vector2 _direction = Vector2.left;
    void Awake()
    {
        _rb = GetComponent<Rigidbody2D>();
        _sr = GetComponent<SpriteRenderer>();
        _col = GetComponent<Collider2D>();
    }
    void Update()
    {
        Vector2 offset = _direction * _col.bounds.extents.x; // distance from center to edge of Collider2D
        Vector2 origin = (Vector2)transform.position + offset;
        var hits=Physics2D.RaycastAll(origin, _direction,_rayCastDistance);
        foreach (var hit in hits){
            if (hit.collider != null && hit.collider.gameObject != gameObject){
                _direction*=-1;
                _sr.flipX = _direction==Vector2.right;
                break;
            }
        }
        _rb.linearVelocity = new Vector2(_direction.x * _speed, _rb.linearVelocity.y);
    }
}
