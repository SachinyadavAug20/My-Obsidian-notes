##### Image 
it is a data structure which stores the pixel data of a image.
- it can be loaded from a file or created programatically
- loaded image -> creates texture, which is a GPU resource that is used to render in `raylib`
- ![[Pasted image 20260723162944.png]]
- Texture 2D is optimized data file used for GPU processing and fast rendering.
```cpp
class Food{
  public:
    Vector2 position={5,6};
    Texture2D texture;
    Food(){
      Image image=LoadImage("assets/food.png");
      texture = LoadTextureFromImage(image);
      UnloadImage(image);
    }
    ~Food(){
      UnloadTexture(texture);
    }
    void Draw(){
      DrawTexture(texture,position.x*cellSize,position.y*cellSize,WHITE);// texture, x,y, color(optional tint color default white)
    }
};
```

### Step 5
snake eat a food -> when head is in same cell as the food.
action to take when snake eats the food
- change position of food to random location
- not pop the last cell(thus increase size of snake)
```cpp
class Food {
public:
  Vector2 position = {5, 6};
  Texture2D texture;
  Food(deque<Vector2> snakeBody) {
    Image image = LoadImage("assets/food.png");
    texture = LoadTextureFromImage(image);
    UnloadImage(image);
    position = GenerateRandomPos(snakeBody);
  }
  ~Food() { UnloadTexture(texture); }
  void Draw() {
    DrawTexture(texture, position.x * cellSize, position.y * cellSize, WHITE);
  }
  Vector2 GenerateRandomCell() {
    float x = GetRandomValue(0, cellCount - 1);
    float y = GetRandomValue(0, cellCount - 1);
    return Vector2{x,y};
  }
  Vector2 GenerateRandomPos(deque<Vector2> body) {
    Vector2 position=GenerateRandomCell();
    while (ElementInDeque(position,body)) {
      position=GenerateRandomCell();
    }
    return position;
  }
};

class Snake {
public:
  deque<Vector2> body = {Vector2{6, 9}, Vector2{5, 9}, Vector2{4, 9}};
  Vector2 direction = {1, 0};
  void draw() {
    for (Vector2 v : body) {
      DrawRectangleRounded(Rectangle{v.x * cellSize, v.y * cellSize, (float)cellSize, (float)cellSize}, 0.7, 6, dark_green);
    }
  }
  void update(Food &food) {
    Vector2 head = body.front();
    if (food.position.x == head.x && food.position.y == head.y)
      food.position = food.GenerateRandomPos(body);
    else
      body.pop_back();
    body.push_front(Vector2{head.x + direction.x, head.y + direction.y});
  }
};

class Game {
public:
  Snake snake = Snake();
  Food food = Food(snake.body);

  void draw() {
    snake.draw();
    food.Draw();
  }
  void update() {
    if (eventTriggered(0.2))
      snake.update(food);
    if (IsKeyPressed(KEY_UP) && snake.direction.y != 1)
      snake.direction = {0, -1};
    else if (IsKeyPressed(KEY_DOWN) && snake.direction.y != -1)
      snake.direction = {0, 1};
    else if (IsKeyPressed(KEY_LEFT) && snake.direction.x != 1)
      snake.direction = {-1, 0};
    else if (IsKeyPressed(KEY_RIGHT) && snake.direction.x != -1)
      snake.direction = {1, 0};
  }
};
```

```cpp
class Game {
public:
  Snake snake = Snake();
  Food food = Food(snake.body);
  bool running=true;

  void draw() {
    snake.draw();
    food.Draw();
  }
  void update() {
    if(running){
      if (eventTriggered(0.2)) snake.update(food);
      checkCollisionWithEdge();
      checkCollisionWithBody();
    }
    snake.snakeMovement(running);
  }
  void checkCollisionWithEdge(){
    if(snake.body[0].x==cellCount || snake.body[0].x==-1 || snake.body[0].y==cellCount || snake.body[0].y==-1) restartGame();
  }
  void checkCollisionWithBody(){
    deque<Vector2> headlessbody=snake.body;
    headlessbody.pop_front();
    if(ElementInDeque(snake.body[0],headlessbody)) restartGame();
  }
  void restartGame(){
    snake.body = {Vector2{6, 9}, Vector2{5, 9}, Vector2{4, 9}};
    food.position=food.GenerateRandomPos(snake.body);
    running=false;
  }
};
```
