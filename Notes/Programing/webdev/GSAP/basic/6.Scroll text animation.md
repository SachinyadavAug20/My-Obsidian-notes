JS
```js
gsap.to("#move",{
  transform:"translateX(-100%)",
  ease:"none",
  duration:2,
  repeat:-1,
})

window.addEventListener("wheel",function(e){
  if(e.deltaY>0){
    gsap.to("#move",{
      transform:"translateX(-200%)",
      ease:"none",
      duration:4,
      repeat:-1,
    })
    gsap.to("#move img",{
      rotate:0,
    })
  }else{
    gsap.to("#move",{
      transform:"translateX(0%)",
      ease:"none",
      duration:4,
      repeat:-1,
    })
    gsap.to("#move img",{
      rotate:180,
    })
  }
})
```
