class Paddle{
  public:
    pair<float,float> cords;
    int speedY;
    float width,height;
    void Draw(){
      DrawRectangle(cords.first,cords.second,width,height,WHITE);

    void update(){
      if(0>cords.second) cords.second=0;
      else if(cords.second>=(GetScreenHeight()-height)) cords.second=GetScreenHeight()-height;

};

class Ball{
  public:
    float x,y;
    int speedX,speedY;
    int radius;
  void Draw(){
    DrawCircle(x,y,radius,WHITE);
  }
  void update(){
    x+=speedX;
    y+=speedY;
    if(y+radius>=GetScreenHeight() || y-radius<=0) speedY*=-1;
    if(x+radius>=GetScreenWidth()){ 
      computerScore++;
      reset();
    }
    if( x-radius<=0){ 
      playerScore++;
      reset();
    }
  }
  void reset(){
    x=GetScreenWidth()/2.0;
    y=GetScreenHeight()/2.0;
    int speed_choice[2]={-1,1};
    speedX*=speed_choice[GetRandomValue(0,1)];
    speedY*=speed_choice[GetRandomValue(0,1)];
  }
};
