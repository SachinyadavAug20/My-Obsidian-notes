ThreeJS need 3 things to setup the scene
- render-er
- camera
- scene object
in `index.js`
```js
import * as THREE from "three";

const w = window.innerWidth;
const h = window.innerHeight;
const renderer = new THREE.WebGLRenderer({ antialias: true }); // make it better by antialias
renderer.setSize(w, h);

// append renderer to DOM
document.body.appendChild(renderer.domElement); // render is a canvas element

const fov = 75;
const aspect = w / h;
const near = 0.1; // all things closer than this are invisible
const far = 10; // all things further than this are invisible
const camera = new THREE.PerspectiveCamera(fov, aspect, near, far);
camera.position.z = 2; // move camera back to see center of scene

const scene = new THREE.scene();

renderer.render(scene, camera);
```

can interact with mouse, can rotate drag zoom etc
can have damping in it.
```js
import * as THREE from "three";
import {OrbitControls} from "jsm/controls/OrbitControls.js"

const w = window.innerWidth;
const h = window.innerHeight;
const renderer = new THREE.WebGLRenderer({ antialias: true }); // make it better by antialias
renderer.setSize(w, h);

// append renderer to DOM
document.body.appendChild(renderer.domElement); // render is a canvas element

const fov = 75;
const aspect = w / h;
const near = 0.1; // all things closer than this are invisible
const far = 10; // all things further than this are invisible
const camera = new THREE.PerspectiveCamera(fov, aspect, near, far);
camera.position.z = 2; // move camera back to see center of scene

const scene = new THREE.Scene();
const geo = new THREE.IcosahedronGeometry(0.7, 2); // size,detail
const mat = new THREE.MeshStandardMaterial({
  color: 0xccffff,
  flatShading:true
});
const hemisphereLight = new THREE.HemisphereLight(0x0001ff, 0xcc5500);
scene.add(hemisphereLight);
const mesh = new THREE.Mesh(geo, mat);
scene.add(mesh);

const wireMat=new THREE.MeshBasicMaterial({
  color:0xffffff,
  wireframe:true
})
const wireMesh=new THREE.Mesh(geo,wireMat)
mesh.add(wireMesh)

const controls = new OrbitControls(camera, renderer.domElement);
controls.enableDamping=true
controls.dampingFactor=0.03;
controls.update();

function animate(t=0) {
  console.log(t);
  requestAnimationFrame(animate);
  mesh.rotateY(0.02);
  mesh.scale.setScalar(Math.sin(t*0.001)*0.2+1.000001);
  renderer.render(scene, camera);
  controls.update();
}
animate()
```
