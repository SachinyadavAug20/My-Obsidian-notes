add `stragger` -ve reverse order and +ve in order
```js
function breakText(text) {
  var h1Text = text.textContent; // MeowMegaCorp

  // split string
  var split = h1Text.split(""); // it give array ["M", "e", "o", "w", "M", "e", "g", "a", "C", "o", "r", "p"]

  var clutter = "";
  const half=(h1Text.length/2);
  split.forEach(function (e,i) {
    if(i<half){
      clutter += `<span class="h1-1st-half">${e}</span>`;
    }else{
      clutter += `<span class="h1-2st-half">${e}</span>`;
    }
  });
  h1.innerHTML = clutter;
}
var h1 = document.querySelector("h1"); // <h1>MeowMegaCorp</h1>
breakText(h1);

// h1 span is display inline : change it to inline-block (make animation)
gsap.from(".h1-1st-half", {
  y:75,
  duration:0.5,
  opacity:0,
  stagger:0.15
})

gsap.from(".h1-2st-half", {
  y:75,
  duration:0.5,
  opacity:0,
  stagger:-0.15
})
```
