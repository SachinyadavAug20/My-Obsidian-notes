## Frog 1
![[Pasted image 20261029191326.png]]
at each step have 2 possible do jump of +1 or +2.
top down => reach stop n from 1.
cost to reach nth stop.
memorize the minimum cost to reach `ith` stone.
#### Step 1
brute for recursion
```cpp
int n;

int jump(int i, vector<int> &a){
    if(i==0) return 0;
    int cost=INT_MAX;
    // way 1
    cost=min(cost,jump(i-1,a)+abs(a[i]-a[i-1]));
    // way 1
    if(i>1) cost=min(cost,jump(i-2,a)+abs(a[i]-a[i-2]));
    return cost;
}

signed main() {
    cin>>n;
    vector<int> a(n);
    for(int i=0;i<n;i++){
        cin>>a[i];
    }
    cout<<jump(n-1, a)<<endl;
    return 0;
}
```
now it can give answer to any test case but, it is O(2^n) which is TLE
#### Step 2
do memorization 
```cpp
const int N=1e5+10;
int n;
int dp[N]; // minimum to reach stop at index i

int jump(int i, vector<int> &a){
    if(i==0) return 0;
    if(dp[i]!=-1) return dp[i];
    int cost=INT_MAX;
    // way 1
    cost=min(cost,jump(i-1,a)+abs(a[i]-a[i-1]));
    // way 1
    if(i>1) cost=min(cost,jump(i-2,a)+abs(a[i]-a[i-2]));
    return dp[i]=cost;
}

signed main() {
    fastio();
    memset(dp,-1,sizeof(dp));
    cin>>n;
    vector<int> a(n);
    for(int i=0;i<n;i++){
        cin>>a[i];
    }
    cout<<jump(n-1, a)<<endl;
    return 0;
}
```
done accepted

## Frog 2
now, if steps can be +1,+2..., +k.
![[Pasted image 20261029204407.png]]
brute force recursion is most important.
```cpp
int n,k;

int jumpk(int i,vector<int> &a){
    if(i==0) return 0;
    int cost=INT_MAX;
    for(int j=1;j<=k;j++){
        if(i-j<0) break;
        cost=min(cost,abs(a[i]-a[i-j])+jumpk(i-j, a));
    }
    return cost;
}

signed main() {
    fastio();
    cin>>n>>k;
    vector<int> a(n);
    for(int i=0;i<n;i++){
        cin>>a[i];
    }
    cout<<jumpk(n-1, a)<<endl;
    return 0;
}
```
it works but it is O(k^n)
now memorize it will become O(k\*n)
```cpp
const int N=1e5+10;
int n,k;
int dp[N];

int jumpk(int i,vector<int> &a){
    if(i==0) return 0;
    if(dp[i]!=-1) return dp[i];
    int cost=INT_MAX;
    for(int j=1;j<=k;j++){
        if(i-j<0) break;
        cost=min(cost,abs(a[i]-a[i-j])+jumpk(i-j, a));
    }
    return dp[i]=cost;
}

signed main() {
    memset(dp,-1,sizeof(dp));
    fastio();
    cin>>n>>k;
    vector<int> a(n);
    for(int i=0;i<n;i++){
        cin>>a[i];
    }

}
```
