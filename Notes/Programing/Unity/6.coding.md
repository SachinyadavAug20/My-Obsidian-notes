player script need to change position of rigid body to move
```cs
using UnityEngine;

public class Player : MonoBehaviour {
    void Start() {
    }

    void Update() {
        var horizontal = Input.GetAxis("Horizontal");
        Rigidbody2D rb=GetComponent<Rigidbody2D>(); // search the game object for this component
        rb.velocity = new Vector2(horizontal, 0f); // vector2D is x,y
    }
}
```

## Jump
`getButtonDown()` -> check if present now, not if it is pressed already
`getButton()` -> check if button is pressed
can continuously move up by holding left click
```cs
        if(Input.GetButton("Fire1")){
            vertical=5; // jump on left click
        }
```
make it hold for certain amount of time
make a member variable for the class -> start with _ as standard
```cs
using UnityEngine;

public class Player : MonoBehaviour
{
    private double _jumpEndTime = 0;
    void Update()
    {
        var horizontal = Input.GetAxis("Horizontal");
        Rigidbody2D rb = GetComponent<Rigidbody2D>();
        var vertical = rb.linearVelocity.y;

        if (Input.GetButtonDown("Fire1"))
        {
            _jumpEndTime = Time.time + 1; // allow jump for 1 seconds
        }
        if (Input.GetButton("Fire1") && Time.time < _jumpEndTime)
        {
            vertical = 5; // jump on left click
        }

        rb.linearVelocity = new Vector2(horizontal, vertical);
    }
}
```
