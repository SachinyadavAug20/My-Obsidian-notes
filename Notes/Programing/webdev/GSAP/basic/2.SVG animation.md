###### In GSAP can also change attribute (which can't be changed by css)
```js
var initial_path=`M 5 100 Q 450 100, 895 100`;

var final_path=`M 5 100 Q 450 100, 895 100`;

 var string=document.querySelector("#string");

string.addEventListener("mousemove", function (dets) {
  const rect = string.getBoundingClientRect();
  path = `M 5 100 Q ${dets.clientX - rect.left} ${dets.clientY - rect.top}, 895 100`;
  gsap.to("svg path", {
    attr: {
      d: path,
    },
    duration: 0.3,
  });
});

// reset
string.addEventListener("mouseleave",function(){
  gsap.to("svg path",{
    attr:{
      d:initial_path
    },
  })
})
```
can have different ease
[Easing | GSAP | Docs & Learning](https://gsap.com/docs/v3/Eases/) use this to make string effect for it.
```js
var initial_path=`M 5 200 Q 450 200, 895 200`;

var final_path=`M 5 200 Q 450 200, 895 200`;

 var string=document.querySelector("#string");

string.addEventListener("mousemove", function (dets) {
  const rect = string.getBoundingClientRect();
  path = `M 5 200 Q ${dets.clientX - rect.left} ${dets.clientY - rect.top}, 895 200`;
  gsap.to("svg path", {
    attr: {
      d: path,
    },
    ease:"power3.out",
    duration: 0.2,
  });
});

// reset
string.addEventListener("mouseleave",function(){
  gsap.to("svg path",{
    attr:{
      d:initial_path
    },
    ease:"elastic.out(2,0.1)", // gritter type
    duration:0.4
  })
})
```
