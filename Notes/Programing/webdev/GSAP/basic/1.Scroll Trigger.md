markers used in.when object's start and end crosses objects triggers animation ==> this us scroll trigger
- object's start and end move as you scroll
- screen's (scroller-end and scroller-start) 
```js
gsap.from("#page1 #box",{
  scale:0,
  duration:1,
  delay:0.5,
  rotate:360,
})

gsap.from("#page2 h1",{
  opacity:0,
  duration:2,
  x:500,
  scale:0,
  rotate:-90,
  scrollTrigger:{
    trigger:"#page2 h1",
    scroller:"body",
    // markers:true,
    start:"top 50%",
    end:"bottom 50%",
    scrub:1,
  }
})
gsap.from("#page2 h2",{
  opacity:0,
  duration:2,
  scale:0,
  rotate:90,
  x:-500,
  scrollTrigger:{
    trigger:"#page2 h2",
    scroller:"body",
    markers:true,
    start:"top 60%",
    end:"bottom 50%",
    scrub:1,
  }
})

gsap.from("#page3 #box",{
  scale:0,
  duration:1,
  delay:0.5,
  rotate:360,
  scrollTrigger:{
    trigger:"#page3 #box", // what to trigger
    scroller:"body",       // scroll in 
    // markers:true,     // marker for development default start at 0%,0%
    start:"top 60%",       // when to start
    end:"bottom 40%",      // when to end 
    scrub:1,               // instead of duration till cross middle of start and end of scoller
  },
})
```
scrub -> animation depend on scroll < true, false/ number> for smoothness
