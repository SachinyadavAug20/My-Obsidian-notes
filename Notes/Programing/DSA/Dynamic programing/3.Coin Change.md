```cpp
class Solution {
public:
    int coinChange(vector<int>& coins, int amount) {
        sort(coins.begin(),coins.end());
        int ans=0;
        for(int i=coins.size()-1;i>=0;i--){
            if(amount<=0) break;
            int cc=amount/coins[i];
            amount-=(coins[i]*cc);
            ans+=cc;
        }
        return amount==0?ans:-1;
    }
};
```

```cpp
class Solution {
public:
    int fun(int amount, vector<int>& coins) {
        if (amount == 0) return 0;
        
        int ans = INT_MAX;
        for (int coin : coins) {
            if (amount - coin >= 0) {
                int res = fun(amount - coin, coins);
                if (res != INT_MAX) {
                    ans = min(ans, res + 1);
                }
            }
        }
        return ans;
    }

    int coinChange(vector<int>& coins, int amount) {
        int ans = fun(amount, coins);
        return (ans == INT_MAX) ? -1 : ans;
    }
};
```

```cpp
class Solution {
public:
    static const int N=1e5+10;
    int dp[N];
    int fun(int amount, vector<int>& coins) {
        if (amount == 0) return 0;
        if(dp[amount]!=-1) return dp[amount];
        
        int ans = INT_MAX;
        for (int coin : coins) {
            if (amount - coin >= 0) {
                int res = fun(amount - coin, coins);
                if (res != INT_MAX) {
                    ans = min(ans, res + 1);
                }
            }
        }
        return dp[amount]=ans;
    }

    int coinChange(vector<int>& coins, int amount) {
        memset(dp,-1,sizeof(dp));
        int ans = fun(amount, coins);
        return (ans == INT_MAX) ? -1 : ans;
    }
};
```

```cpp
class Solution {

public:
    static const int N=1e5+10;
    int dp[N];
    int fun(int amount, vector<int>& coins) {
        if (amount == 0){
            return 1;// 1 way
        } 
        if(dp[amount]!=-1){
            return dp[amount];
        }
        
        int ways = INT_MAX;
        for (int coin : coins) {
            if (amount - coin >= 0) {
                ways += fun(amount - coin, coins);
            }
        }
        return dp[amount]=ways;
    }

    int change(int amount,vector<int>& coins) {
        memset(dp,-1,sizeof(dp));
        return fun(amount,coins);
    }
};
```
problem is to maintain order to avoid combination
```cpp
class Solution {
    // obs
    // will count using same => 
    // problem : give all permutation need only combination
    // solve: maintain order 
public:
    static const int N=1e5+10;
    static const int M=310;
    int dp[N][M];
    int fun(int index,int amount, vector<int>& coins) {
        if (amount == 0) return 1;
        if(dp[amount][index]!=-1){
            return dp[amount][index];
        }
        
        int ways = 0;
        for(int coin_amount=0;coin_amount<amount;coin_amount+=coins[index]){
            ways += fun(index-1,amount - coin_amount, coins);
        }
        return dp[amount][index]=ways;
    }

    int change(int amount,vector<int>& coins) {
        memset(dp,-1,sizeof(dp));
        return fun(0,amount,coins);
    }
};
```
