```cpp
class Solution {
public:
    int coinChange(vector<int>& coins, int amount) {
        sort(coins.begin(),coins.end());
        int ans=0;
        for(int i=coins.size()-1;i>=0;i--){
            if(amount<=0) break;
            int cc=amount/coins[i];
            amount-=(coins[i]*cc);
            ans+=cc;
        }
        return amount==0?ans:-1;
    }
};
```

do brute force recursion.

```cpp
class Solution {
public:
    int fun(int amount, vector<int>& coins) {
        if (amount == 0) return 0;
        
        int ans = INT_MAX;
        for (int coin : coins) {
            if (amount - coin >= 0) {
                int res = fun(amount - coin, coins);
                if (res != INT_MAX) {
                    ans = min(ans, res + 1);
                }
            }
        }
        return ans;
    }

    int coinChange(vector<int>& coins, int amount) {
        int ans = fun(amount, coins);
        return (ans == INT_MAX) ? -1 : ans;
    }
};
```

```cpp
class Solution {
public:
    static const int N=1e5+10;
    int dp[N];
    int fun(int amount, vector<int>& coins) {
        if (amount == 0) return 0;
        if(dp[amount]!=-1) return dp[amount];
        
        int ans = INT_MAX;
        for (int coin : coins) {
            if (amount - coin >= 0) {
                int res = fun(amount - coin, coins);
                if (res != INT_MAX) {
                    ans = min(ans, res + 1);
                }
            }
        }
        return dp[amount]=ans;
    }

    int coinChange(vector<int>& coins, int amount) {
        memset(dp,-1,sizeof(dp));
        int ans = fun(amount, coins);
        return (ans == INT_MAX) ? -1 : ans;
    }
};
```
