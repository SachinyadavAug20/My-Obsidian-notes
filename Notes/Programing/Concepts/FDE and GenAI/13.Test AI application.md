## AI Evals
An AI evaluation or eval is a systematic way of measuring whether an AI system behaves the way we want.
It is like test cases.
expected output is behavior:(like mention terms)
![[Pasted image 20261026124405.png]]
there are 3 things to test AI application
- Test input (that is user prompt)
- Expected behavior 
	- what should be mentioned, proper tool calling etc.
- Evaluation mechanism 
	- can't do direct string matching as have non-deterministic behavior.
	- there are many ways to evaluate.
	- create situation and states which can be tested using code.
	- LLM-as-a-judge or human-as-a-judge

```java
@Component
public class DeliveryEstimateTool {

    private final ToolTracker toolTracker;

    public DeliveryEstimateTool(ToolTracker toolTracker) {
        this.toolTracker = toolTracker;
    }

    @Tool(description = """
            Get the expected delivery date for an order.
            Use this when the user asks when the order will arrive,
            when it will be delivered, or asks for a delivery estimate.
           """
    )
    public String getDeliveryEstimate(
            @ToolParam(description ="Order ID such as ORD-101") String orderId) {

        String normalizedOrderId = orderId.trim().toUpperCase();
        String result;

        if ("ORD-101".equalsIgnoreCase(normalizedOrderId)) {
            result = "Order ORD-101 is expected to be delivered on 15 October 2026.";
        }

        else if ("ORD-102".equalsIgnoreCase(normalizedOrderId)) {
            result = "Order ORD-102 has already been delivered.";
        }

        else if ("ORD-103".equalsIgnoreCase(normalizedOrderId)) {
            result = "Order ORD-103 has already been delivered.";
        }
        else {
            result = "Delivery information not found.";
        }

        toolTracker.record("getDeliveryEstimate", normalizedOrderId, result);

        return result;
    }
}
```

tool tracker
```java
@Component
public class ToolTracker {

    private final List<String> toolsUsed = new ArrayList<String>();
    private final Map<String, String> toolArguments = new LinkedHashMap<String, String>();
    private final Map<String, String> toolResults = new LinkedHashMap<String, String>();

    public void record(String toolName, String argument, String result) {
        if (!toolsUsed.contains(toolName)) {
            toolsUsed.add(toolName);
        }
        toolArguments.put(toolName, argument);
        toolResults.put(toolName, result);
    }

    public void clear() {
        toolsUsed.clear();
        toolArguments.clear();
        toolResults.clear();
    }

    public List<String> getToolsUsed() {
        return new ArrayList<String>(toolsUsed);
    }

    public Map<String, String> getToolArguments() {
        return new LinkedHashMap<String, String>(toolArguments);
    }

    public Map<String, String> getToolResults() {
        return new LinkedHashMap<String, String>(toolResults);
    }
}
```

```java
@Service
public class SupportService {

    private final ChatClient chatClient;
    private final VectorStore vectorStore;
    private final OrderStatusTool orderStatusTool;
    private final DeliveryEstimateTool deliveryEstimateTool;
    private final RefundAmountTool refundAmountTool;
    private final ToolTracker toolTracker;

    @Value("classpath*:knowledge/*.txt")
    private Resource[] policyFiles;

    public SupportService(
            ChatClient.Builder builder,
            VectorStore vectorStore,
            OrderStatusTool orderStatusTool,
            DeliveryEstimateTool deliveryEstimateTool,
            RefundAmountTool refundAmountTool,
            ToolTracker toolTracker) {

        this.chatClient = builder.build();
        this.vectorStore = vectorStore;
        this.orderStatusTool = orderStatusTool;
        this.deliveryEstimateTool = deliveryEstimateTool;
        this.refundAmountTool = refundAmountTool;
        this.toolTracker = toolTracker;
    }

    // Load our three policy files into the in-memory vector store when the application starts.
    @PostConstruct
    public void loadKnowledgeBase() throws IOException {
        List<Document> documents = new ArrayList<Document>();

        for (Resource resource : policyFiles) {

            String content = StreamUtils.copyToString(resource.getInputStream(), StandardCharsets.UTF_8);
            String filename = resource.getFilename();

            Map<String, Object> metadata = new HashMap<String, Object>();
            metadata.put("source", filename);

            Document document = new Document(content, metadata);
            documents.add(document);
        }

        vectorStore.add(documents);
        System.out.println("Knowledge base loaded: " + documents.size() + " documents");
    }

    public AgentResponse chat(String question) {

        // Every request starts with a fresh trace. Later our evals will inspect this.
        toolTracker.clear();

        // Retrieve the most relevant company policy.
        List<Document> documents = vectorStore.similaritySearch(SearchRequest.builder()
                .query(question)
                .topK(1)
                .build()
        );

        // Convert retrieved documents into context for the LLM.

        StringBuilder contextBuilder = new StringBuilder();
        List<String> retrievedDocuments = new ArrayList<String>();

        for (Document document : documents) {
            String source = String.valueOf(document.getMetadata().get("source"));
            retrievedDocuments.add(source);
            contextBuilder.append("SOURCE: ").append(source).append("\n");
            contextBuilder.append(document.getText()).append("\n\n");
        }
        String context = contextBuilder.toString();

        // Give the LLM: policy context, user question, available tools
        String systemPrompt =
                "You are an e-commerce customer support agent.\n\n" +

                        "Use the supplied POLICY CONTEXT for questions about company policies.\n\n" +

                        "Use tools only when the user asks for information about a " +
                        "specific order and provides an order ID such as ORD-101.\n\n" +

                        "Do not call an order tool for general policy questions.\n\n" +

                        "If the user already provides enough information in the question, " +
                        "use that information directly instead of calling a tool.\n\n" +

                        "For example, if the user says an item was received three weeks ago, " +
                        "treat that as known information and compare it with the return policy. " +
                        "Do not call getOrderStatus unless an order ID is provided and " +
                        "order-specific information is actually required.\n\n" +

                        "Tool selection rules:\n" +

                        "1. Current status of a specific order -> getOrderStatus\n" +
                        "2. Expected delivery date of a specific order -> getDeliveryEstimate\n" +
                        "3. Refund amount for a specific order -> getRefundAmount\n\n" +

                        "If return eligibility is asked for a specific order ID, " +
                        "and delivery information is not already provided by the user, " +
                        "use getOrderStatus and combine the result with the return policy.\n\n" +

                        "Never guess order-specific information.\n\n" +

                        "The retrieved policy context may sometimes be unrelated. " +
                        "Ignore irrelevant policy information.\n\n" +

                        "POLICY CONTEXT:\n" +
                        context;

        // Ask the model. Spring AI handles the tool-call loop automatically.
        String answer = chatClient.prompt()
                .system(systemPrompt)
                .user(question)
                .tools(orderStatusTool, deliveryEstimateTool, refundAmountTool)
                .call()
                .content();

        // Return both: answer for user & trace for evaluation
        return new AgentResponse(
                answer,
                retrievedDocuments,
                toolTracker.getToolsUsed(),
                toolTracker.getToolArguments(),
                toolTracker.getToolResults(),
                context
        );
    }
}
```

```java
@SpringBootTest
public class SupportAgentEvalTest {

    @Autowired
    private SupportService supportService;

    @Autowired
    private ChatClient.Builder chatClientBuilder;// same LLM used a judge with different system prompt

    @Autowired
    private JsonMapper jsonMapper; // JSON response map to model class

    @Test
    public void runGoldenDataset() throws Exception {

        // Load Golden DataSet
        InputStream inputStream = new ClassPathResource("golden-dataset.json").getInputStream();
        List<EvalCase> evalCases = jsonMapper.readValue(inputStream, new TypeReference<List<EvalCase>>() {});
        inputStream.close();

        // Separate ChatClient for our judge.
        ChatClient judgeClient = chatClientBuilder.build();

        // Store failed cases.
        List<String> failures = new ArrayList<String>();

        int relevanceTotal = 0;
        int correctnessTotal = 0;
        int faithfulnessTotal = 0;
        int helpfulnessTotal = 0;
        int hallucinationFreeTotal = 0;

        // STEP 2: Run every golden test.
        for (EvalCase evalCase : evalCases) {

            System.out.println();
            System.out.println("========================================");
            System.out.println("TEST: " + evalCase.getId());
            System.out.println("QUESTION: " + evalCase.getQuestion());

            // STEP 3: Run actual AI application
            AgentResponse response = supportService.chat(evalCase.getQuestion());

            // STEP 4: Deterministic evaluations.
            boolean responsePass = response.getAnswer() != null
                    &&
                    !response.getAnswer().trim().isEmpty();

            boolean toolPass;

            if (evalCase.getExpectedTool() == null) {
                toolPass = response.getToolsUsed().isEmpty();
            } else {
                toolPass = response.getToolsUsed().contains(evalCase.getExpectedTool());
            }

            boolean argumentPass = true;
            
            if (evalCase.getExpectedTool() != null) {
                String actualArgument = response.getToolArguments().get(evalCase.getExpectedTool());

                argumentPass = evalCase.getExpectedToolArgument().equals(actualArgument);
            }

            boolean sourcePass = true;

            if (evalCase.getExpectedSource() != null) {
                sourcePass = response.getRetrievedDocuments().contains(evalCase.getExpectedSource());
            }

            boolean deterministicPass = responsePass && toolPass && argumentPass && sourcePass;

            // STEP 5: Quality evaluation.
            JudgeResult judgeResult = judge(judgeClient, evalCase, response);

            // We choose 4/5 as our demo quality threshold.
            boolean qualityPass = judgeResult.getRelevance() >= 4 &&
                    judgeResult.getCorrectness() >= 4 &&
                    judgeResult.getFaithfulness() >= 4 &&
                    judgeResult.getHelpfulness() >= 4 &&
                    judgeResult.getHallucinationFree() >= 4;


            boolean finalPass = deterministicPass && qualityPass;

            // STEP 6: Print report.
            System.out.println();
            System.out.println("DETERMINISTIC EVALS");
            System.out.println("Response exists: " + passFail(responsePass));
            System.out.println("Correct tool: " + passFail(toolPass));
            System.out.println("Correct tool argument: " + passFail(argumentPass));
            System.out.println("Expected source retrieved: " + passFail(sourcePass));
            System.out.println();
            System.out.println("QUALITY EVALS");
            System.out.println("Relevance: " + judgeResult.getRelevance() + "/5");
            System.out.println("Correctness: " + judgeResult.getCorrectness() + "/5");
            System.out.println("Faithfulness: " + judgeResult.getFaithfulness() + "/5");
            System.out.println("Helpfulness: " + judgeResult.getHelpfulness() + "/5");
            System.out.println("Hallucination Free: " + judgeResult.getHallucinationFree() + "/5");
            System.out.println("Judge Reason: " + judgeResult.getReason());
            System.out.println();
            System.out.println("FINAL: " + passFail(finalPass));

            // STEP 7: Aggregate scores.
            relevanceTotal += judgeResult.getRelevance();
            correctnessTotal += judgeResult.getCorrectness();
            faithfulnessTotal += judgeResult.getFaithfulness();
            helpfulnessTotal += judgeResult.getHelpfulness();
            hallucinationFreeTotal += judgeResult.getHallucinationFree();

            if (!finalPass) {
                failures.add(evalCase.getId());
            }
        }

        // STEP 8: Overall report.
        int totalTests = evalCases.size();

        System.out.println();
        System.out.println();
        System.out.println("========================================");
        System.out.println("FINAL EVALUATION REPORT");
        System.out.println("========================================");
        System.out.println("Total Tests: " + totalTests);
        System.out.println("Passed: " + (totalTests - failures.size()));
        System.out.println("Failed: " + failures.size());
        System.out.println();
        System.out.println("Average Relevance: " + average(relevanceTotal, totalTests));
        System.out.println("Average Correctness: " + average(correctnessTotal, totalTests));
        System.out.println("Average Faithfulness: " + average(faithfulnessTotal, totalTests));
        System.out.println("Average Helpfulness: " + average(helpfulnessTotal, totalTests));
        System.out.println("Average Hallucination Free: " + average(hallucinationFreeTotal, totalTests));


        // STEP 9: Finally fail the JUnit test
        Assertions.assertTrue(failures.isEmpty(), "Failed Eval Cases: " + failures);
    }

    private JudgeResult judge(ChatClient judgeClient, EvalCase evalCase, AgentResponse response) {
        String prompt = "Evaluate the AI answer below.\n\n" +

                "USER QUESTION:\n" +
                evalCase.getQuestion() +
                "\n\n" +

                "EXPECTED ANSWER:\n" +
                evalCase.getExpectedAnswer() +
                "\n\n" +

                "EXPECTED BEHAVIOR:\n" +
                evalCase.getExpectedBehavior() +
                "\n\n" +

                "RETRIEVED POLICY CONTEXT:\n" +
                response.getContext() +
                "\n\n" +

                "TOOL RESULTS:\n" +
                response.getToolResults().toString() +
                "\n\n" +

                "ACTUAL ANSWER:\n" +
                response.getAnswer() +
                "\n\n" +

                "Score the answer from 1 to 5 " +
                "for each metric.\n\n" +

                "RELEVANCE:\n" +
                "5 = directly answers the question.\n" +
                "4 = mostly answers it.\n" +
                "3 = partially answers it.\n" +
                "2 = barely related.\n" +
                "1 = irrelevant.\n\n" +

                "CORRECTNESS:\n" +
                "5 = fully consistent with expected answer and behavior.\n" +
                "4 = mostly correct with a minor issue.\n" +
                "3 = partially correct.\n" +
                "2 = major errors.\n" +
                "1 = substantially incorrect.\n\n" +

                "FAITHFULNESS:\n" +
                "5 = every factual claim is supported by policy context " +
                "or tool results.\n" +
                "4 = almost fully supported.\n" +
                "3 = meaningful unsupported claims exist.\n" +
                "2 = large parts are unsupported.\n" +
                "1 = contradicts or ignores the evidence.\n\n" +

                "HELPFULNESS:\n" +
                "5 = clear, complete and useful.\n" +
                "4 = useful with a minor limitation.\n" +
                "3 = somewhat useful.\n" +
                "2 = minimally useful.\n" +
                "1 = not useful.\n\n" +

                "HALLUCINATION FREE:\n" +
                "5 = no fabricated or unsupported claims.\n" +
                "4 = insignificant unsupported detail.\n" +
                "3 = some unsupported information.\n" +
                "2 = substantial unsupported information.\n" +
                "1 = heavily fabricated.\n\n" +

                "Use ONLY the expected answer, expected behavior, " +
                "policy context and tool results as evidence. " +
                "Do not use outside knowledge.";


        return judgeClient.prompt()
                .system("You are a strict evaluator of AI application responses.")
                .user(prompt)
                .call()
                .entity(JudgeResult.class);
    }

    private String passFail(boolean value) {
        return value ? "PASS" : "FAIL";
    }

    private double average(int total, int count) {
        return Math.round(((double) total / count) * 100.0) / 100.0;
    }
```
