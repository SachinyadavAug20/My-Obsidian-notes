also cache sprite component as used every-where `Update` which is called every frame
as getting component is slow/expensive operation
```cs
using UnityEngine;

public class Player : MonoBehaviour
{
    float _jumpEndTime = 0f;
    [SerializeField] float _jumpVelocity = 5f;
    [SerializeField] float _horizontalVectocity = 3;
    [SerializeField] float _jumpTime = 0.55f;
    [SerializeField] Sprite _jumpSprite;
    [SerializeField] Sprite _defaultSprite;
    public bool isGrounded;
    SpriteRenderer _sr;
    private void Awake(){
        _sr = GetComponent<SpriteRenderer>();
        _defaultSprite=_sr.sprite;
    }

    void OnDrawGizmos() {
        SpriteRenderer sr = GetComponent<SpriteRenderer>();
        Vector2 origin = new Vector2(transform.position.x, transform.position.y - sr.bounds.extents.y);
        Gizmos.color = Color.red;
        Gizmos.DrawLine(origin, origin + Vector2.down * 0.1f); // arrow 0.1m below
    }
    void Update()
    {
        Vector2 origin = new Vector2(transform.position.x, transform.position.y - _sr.bounds.extents.y);
        var hit = Physics2D.Raycast(origin, Vector2.down, 0.1f); // start,direction,length
        if (hit.collider)
        { // if hit something not null/air
            isGrounded = true;
            _sr.sprite = _defaultSprite;
        }
        else
        {
            isGrounded = false;
            _sr.sprite = _jumpSprite;
        }
        var horizontal = Input.GetAxis("Horizontal") * _horizontalVectocity;
        Rigidbody2D rb = GetComponent<Rigidbody2D>();
        var vertical = rb.linearVelocity.y;

        if (Input.GetButtonDown("Fire1") && isGrounded) // IMP as start restruction
        {
            _jumpEndTime = Time.time + _jumpTime;
        }
        if (Input.GetButton("Fire1") && Time.time < _jumpEndTime)
        {
            vertical = _jumpVelocity;
        }
        rb.linearVelocity = new Vector2(horizontal, vertical);
    }
}
```
for `OnDrawGizmos()` can't cache because object live cycle => it may be called before awake method
[Unity - Manual: Event function execution order](https://docs.unity3d.com/6000.0/Documentation/Manual/execution-order.html)
