## Game loop
![[Pasted image 20260803163728.png]]
basic code
```cpp
const int windowWidth=1280;
const int windowHeight=720;
int main(){
  InitWindow(windowWidth, windowHeight, "Swarm");
  SetTargetFPS(60);
  ChangeDirectory(TextFormat("%s../assets/images/",GetApplicationDirectory()));
  Texture2D background = LoadTexture("Floor.png");
  Texture2D walls= LoadTexture("Walls.png");
  while (!WindowShouldClose()) {
    BeginDrawing();
    ClearBackground(BLACK);
    DrawTexture(background, 0, 0, WHITE);
    DrawTexture(walls, 0, 0, WHITE);
    DrawText("Doom Dam", 10, 10, 20, WHITE);
    EndDrawing();
  }
  UnloadTexture(background);
  CloseWindow();
  return 0;
}
```

code
```cpp
constexpr int BASE_W = 1280;
constexpr int BASE_H = 720;
int main() {
  SetConfigFlags(FLAG_WINDOW_RESIZABLE);
  InitWindow(BASE_W, BASE_H, "Swarm");
  SetTargetFPS(60);

  ChangeDirectory(TextFormat("%s../assets/images/", GetApplicationDirectory()));
  Texture2D background = LoadTexture("Floor.png");
  Texture2D walls = LoadTexture("Walls.png");

  RenderTexture2D canvas=LoadRenderTexture(BASE_W, BASE_H);

  while (!WindowShouldClose()) {
    BeginTextureMode(canvas);
      DrawTexture(background, 0, 0, WHITE);
      DrawTexture(walls, 0, 0, WHITE);
      DrawText("Doom Dam", 24, 24, 25, WHITE);
    EndTextureMode();

    float scale = min((float)GetScreenWidth() / BASE_W, (float)GetScreenHeight() / BASE_H);
    float offsetX = (GetScreenWidth() - BASE_W * scale) *0.5f;
    float offsetY = (GetScreenHeight() - BASE_H * scale) *0.5f;

    Rectangle srcRect = { 0, 0, (float)BASE_W, -(float)BASE_H };
    Rectangle dstRect = { offsetX, offsetY, BASE_W * scale, BASE_H * scale };

    BeginDrawing();
      ClearBackground(BLACK);
      DrawTexturePro(canvas.texture, srcRect, dstRect, {0, 0}, 0, WHITE);
    EndDrawing();
  }
  UnloadTexture(background);
  UnloadTexture(walls);
  UnloadRenderTexture(canvas);
  CloseWindow();
  return 0;
}
```
Reason for -ve sign in `srcRect`'s height
- `DrawText("Doom Dam", 24, 24, ...)` draws into the `RenderTexture2D` canvas. In OpenGL, textures have their origin at the bottom-left, so `RenderTexture2D` content is stored upside-down relative to screen coordinates. When you display it via `DrawTexturePro` without a flip, the whole image is rendered vertically mirrored — so the "top-left" of the canvas shows up at the bottom-left of the window.
- The text is actually at the top-left of the canvas; it just appears at the bottom-left on screen.
