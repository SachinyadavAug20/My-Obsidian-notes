
in properties
```properties
spring.datasource.url=jdbc:mysql://localhost:3306/student_db
spring.datasource.username=root
spring.datasource.password=root123

spring.jpa.hibernate.ddl-auto=update
```
ddl -> data defination language.(command to create,delete tables)
ddl-auto
- create -> create new table for all entities and drop old table if exist
- update -> if no table then make table else use existing table.(update if schema changes)
- none -> don't do any ddl commands
- validate -> schema matching.
also add properties to see which SQL query it executed and formated.
```properties
spring.jpa.show-sql=true
spring.jpa.properties.hibernate.format_sql=true
```

for repository
```java
@Repository
public class StudentRepository {
    @PersistenceContext
    private EntityManager entityManager;

    public Student saveUser(Student student) {
        entityManager.persist(student);
        return student;
    }
    public Student getUserById(Long id) {
        return entityManager.find(Student.class, id);
    }
    public List<Student> getAllUser() {
        return entityManager.createQuery("SELECT s FROM Student s", Student.class).getResultList();
    }
    public void deleteUser(Student student) {
        entityManager.remove(student);
    }
}
```

make update and delete in service
```java
@Service
public class StudentService {
    private StudentRepository studentRepository;

    StudentService(StudentRepository studentRepository) {
        this.studentRepository = studentRepository;
    }

    @Transactional // will do as transaction(not needed but did to show)
    public Student SaveUser(Student student) {
        return studentRepository.saveUser(student);
    }

    public List<Student> getAllUser() {
        return studentRepository.getAllUser();
    }

    public Student getUserById(Long id) {
        return studentRepository.getUserById(id);
    }

    @Transactional
    public Student updateUser(Student student, Long id) {
        Student s1 = studentRepository.getUserById(id);
        if (s1 == null) throw new RuntimeException("User not found");
        s1.setName(student.getName());
        s1.setAge(student.getAge());
        s1.setEmail(student.getEmail());
        // changed referenc so will update it.
        return s1;
    }

    public void deleteUser(Long id) {
        Student s1 = studentRepository.getUserById(id);
        if (s1 == null) throw new RuntimeException("User not found");
        studentRepository.deleteUser(s1);
    }
}
```
