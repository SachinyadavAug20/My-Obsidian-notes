can control when to start timeline, pause and reverse it.
by using functions like `tl.pause()`,`tl.play`
```js
const menu=document.querySelector("#nav i")
const cross=document.querySelector("#full i")

var tl=gsap.timeline()

tl.to("#full",{
  right:0,
  duration:0.4,
  ease:"power4.out"
})
tl.from("#full h4",{
  x:150,
  opacity:0,
  duration:0.2,
  stagger:0.11,
  ease:"power4.out"
})
tl.from("#full i",{
  opacity:0,
  ease:"power4.out"
})

tl.pause()
menu.addEventListener("click",()=>{tl.play()})
cross.addEventListener("click",()=>{tl.reverse()})
```
