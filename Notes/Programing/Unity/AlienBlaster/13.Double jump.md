```cs
        if (Input.GetButtonDown("Fire1") && _jumpsAvailable > 0) {
            _jumpEndTime = Time.time + _jumpTime;
            _jumpsAvailable--;
            _audioSource.pitch = (_jumpsAvailable > 0) ? 1 : 1.2f;
```
